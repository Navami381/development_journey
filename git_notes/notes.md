### command for adding user
'git config --global user.name "yourname"'

### command for adding user's email
'git config --global user.email "youremail"'

### command for list config details
'git config --list'

### command for initializing git
'git init'

### command for adding all changes to staged area
'git add -A'

## command for adding staged files to local repository
'git commit -m "commit message"'

## command for working tree clean or list all untracked files
'git status'
